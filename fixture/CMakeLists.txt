cmake_minimum_required(VERSION 4.0)
project(profile_fixture LANGUAGES C CXX)
enable_testing()

add_library(fixture_c STATIC src/checksum.c)
add_executable(fixture_test src/main.cpp)
target_link_libraries(fixture_test PRIVATE fixture_c)
if(NOT CMAKE_DISABLE_PRECOMPILE_HEADERS)
  target_precompile_headers(fixture_test PRIVATE <vector> <string>)
endif()
if(PROFILE_REFLECTION STREQUAL "refl_on")
  target_sources(fixture_test PRIVATE src/reflection.cpp)
  target_compile_definitions(fixture_test PRIVATE FIXTURE_REFLECTION=1)
endif()
if(PROFILE_IMPORT_STD STREQUAL "importstd_on")
  target_sources(fixture_test PRIVATE src/import_std.cpp)
  target_compile_definitions(fixture_test PRIVATE FIXTURE_IMPORT_STD=1)
endif()
add_test(NAME fixture_test COMMAND fixture_test)
